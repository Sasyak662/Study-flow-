name: Build Android app

on:
  push:
    paths:
      - 'android/**'
      - '.github/workflows/android.yml'
  workflow_dispatch:

permissions:
  contents: write

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: '17'

      - uses: android-actions/setup-android@v3

      - uses: gradle/actions/setup-gradle@v4
        with:
          gradle-version: '8.9'

      - name: Build the APK
        run: gradle -p android assembleRelease --no-daemon

      - name: Put StudyFlow.apk on the website
        run: |
          cp android/app/build/outputs/apk/release/app-release.apk StudyFlow.apk
          git config user.name "StudyFlow bot"
          git config user.email "actions@users.noreply.github.com"
          git add StudyFlow.apk
          git commit -m "Update StudyFlow Android app" || echo "No change"
          git pull --rebase
          git push

      - uses: actions/upload-artifact@v4
        with:
          name: StudyFlow-android
          path: StudyFlow.apk
